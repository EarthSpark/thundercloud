{{ _('What is Parquet?') }} {{ _('These files are stored in Apache Parquet format, a columnar storage format optimized for analytics.') }} {{ _('Learn how to read and query Parquet files') }}

{{ _('Reading Parquet Files with DuckDB') }}

{{ _('DuckDB is a fast in-process SQL database that can query Parquet files directly. Install it from') }} duckdb.org

{{ _('Readings Data Schema') }}

{{ _('Each readings file contains meter data with columns including:') }}

  • serial - Meter serial number
  • heartbeat_start, heartbeat_end - Reading time window
  • kilowatt_hours - Energy consumed during the period
  • voltage_avg, voltage_min, voltage_max - Voltage measurements
  • current_avg, current_min, current_max - Current measurements
  • power_factor_avg, true_power_avg - Power metrics
  • state - Meter state (integer): 0=Off, 1=On, 3=Error, 6=Throttle, 13=Tamper
  • acct_credit, acct_debt, cost - Account balance info
{{ _('Example Queries') }}
# View first 10 readings with key columns
duckdb -c "SELECT serial, heartbeat_start, kilowatt_hours, voltage_avg, state
           FROM 'your_file.parquet' LIMIT 10"

# Count readings per meter
duckdb -c "SELECT serial, COUNT(*) as reading_count
           FROM 'your_file.parquet'
           GROUP BY serial ORDER BY reading_count DESC"

# Total energy consumption by meter
duckdb -c "SELECT serial, SUM(kilowatt_hours) as total_kwh, AVG(voltage_avg) as avg_voltage
           FROM 'your_file.parquet'
           GROUP BY serial"

# Filter readings where meter was on (state=1) with good voltage
duckdb -c "SELECT * FROM 'your_file.parquet'
           WHERE state = 1 AND voltage_avg > 200"

# Count readings by state
duckdb -c "SELECT state, COUNT(*) as count
           FROM 'your_file.parquet'
           GROUP BY state ORDER BY state"

# Export filtered data to CSV
duckdb -c "COPY (SELECT serial, heartbeat_start, kilowatt_hours, voltage_avg, current_avg, state
           FROM 'your_file.parquet')
           TO 'readings.csv' (HEADER, DELIMITER ',')"
